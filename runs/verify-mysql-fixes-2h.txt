  bash runs/mysql-2h-envelope.sh run ddlsmith-baseline <log> -- \

  bash runs/mysql-2h-envelope.sh run ddlsmith-fixed <log> -- \
      bash runs/verify-mysql-fixes.sh --jar <fixed jar> --parallel 5
